cmake_minimum_required(VERSION 3.20)
project(Evolve VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/EHsc)
endif()
add_library(evolve_core src/core/Project.cpp src/core/Geometry.cpp src/voice/VoiceControl.cpp src/voice/ConversationalProvider.cpp)
target_include_directories(evolve_core PUBLIC src)
add_library(evolve_presentation src/presentation/SceneBridge.cpp)
target_link_libraries(evolve_presentation PUBLIC evolve_core)
find_package(Threads REQUIRED)
target_link_libraries(evolve_core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(evolve_core PRIVATE winhttp)
endif()
if(MSVC)
  target_compile_options(evolve_core PRIVATE /W4 /permissive-)
  target_compile_options(evolve_presentation PRIVATE /W4 /permissive-)
else()
  target_compile_options(evolve_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(evolve_presentation PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(evolve_tests tests/CoreTests.cpp)
  target_link_libraries(evolve_tests PRIVATE evolve_core)
  add_test(NAME evolve_core COMMAND evolve_tests)
  add_executable(evolve_voice_tests tests/VoiceTests.cpp)
  target_link_libraries(evolve_voice_tests PRIVATE evolve_core)
  add_test(NAME evolve_voice COMMAND evolve_voice_tests)
  add_executable(evolve_provider_tests tests/ProviderTests.cpp)
  target_link_libraries(evolve_provider_tests PRIVATE evolve_core)
  add_test(NAME evolve_provider COMMAND evolve_provider_tests)
  add_executable(evolve_scene_bridge_tests tests/SceneBridgeTests.cpp)
  target_link_libraries(evolve_scene_bridge_tests PRIVATE evolve_presentation)
  add_test(NAME evolve_scene_bridge COMMAND evolve_scene_bridge_tests)
endif()
add_executable(evolve_cli legacy/simulation.cpp)
if(WIN32)
  add_executable(Evolve WIN32 src/desktop/Main.cpp src/desktop/VoicePanel.cpp src/voice/WindowsVoice.cpp src/renderer/Dx12Renderer.cpp)
  target_compile_definitions(Evolve PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(Evolve PRIVATE evolve_core d3d12 dxgi d3dcompiler comctl32 comdlg32 shell32 sapi ole32 uuid)
  if(MSVC)
    target_compile_options(Evolve PRIVATE /W4 /permissive-)
    set_property(TARGET Evolve evolve_core evolve_presentation evolve_cli PROPERTY
      MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    if(BUILD_TESTING)
      set_property(TARGET evolve_tests evolve_voice_tests evolve_provider_tests evolve_scene_bridge_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
  endif()
  install(TARGETS Evolve RUNTIME DESTINATION .)
  install(FILES README.md CHANGELOG.md DESTINATION .)
  install(DIRECTORY examples docs DESTINATION .)
endif()
